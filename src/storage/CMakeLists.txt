cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(xboxllr_storage LANGUAGES C CXX)
endif()
if(WIN32)
  add_executable(xboxllr_storage_bridge_test ../../tests/test_native_storage_bridge.c)
  set_property(TARGET xboxllr_storage_bridge_test PROPERTY C_STANDARD 11)
  target_link_libraries(xboxllr_storage_bridge_test PRIVATE xboxllr_storage bcrypt)
  if(MSVC)
    target_compile_definitions(xboxllr_storage_bridge_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(xboxllr_storage SHARED disk_image.cpp fatx_volume.cpp storage_api.cpp)
target_compile_definitions(xboxllr_storage PRIVATE XLLR_STORAGE_SHARED)
target_include_directories(xboxllr_storage PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(xboxllr_ram_cache_test ../../tests/test_ram_cache.cpp)
target_link_libraries(xboxllr_ram_cache_test PRIVATE xboxllr_storage)
enable_testing()
add_test(NAME storage_ram_cache COMMAND xboxllr_ram_cache_test)
if(WIN32)
  add_test(NAME storage_ram_cache_bridge COMMAND xboxllr_storage_bridge_test --ram)
endif()
if(MSVC)
  target_compile_options(xboxllr_storage PRIVATE /W4 /WX /EHsc)
  target_compile_definitions(xboxllr_storage PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(xboxllr_storage PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(xboxllr_disk_probe disk_probe.cpp)
# The probe deliberately calls the C++ implementation; this separate static
# target keeps the DLL's public boundary strictly C.
add_library(xboxllr_storage_core STATIC disk_image.cpp fatx_volume.cpp)
set_target_properties(xboxllr_storage_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(xboxllr_storage_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_definitions(xboxllr_storage_core PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
target_link_libraries(xboxllr_disk_probe PRIVATE xboxllr_storage_core)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  foreach(kind security dvd)
    add_executable(storage_ide_${kind} ../../tests/test_ide_${kind}.cpp ../machine/ide.cpp)
    target_include_directories(storage_ide_${kind} PRIVATE ..)
    target_link_libraries(storage_ide_${kind} PRIVATE xboxllr_storage_core)
    if(kind STREQUAL "dvd")
      target_sources(storage_ide_${kind} PRIVATE ../machine/smbus.cpp)
    endif()
    add_test(NAME storage_ide_${kind} COMMAND storage_ide_${kind})
  endforeach()
endif()
