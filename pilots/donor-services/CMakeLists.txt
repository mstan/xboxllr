cmake_minimum_required(VERSION 3.20)
project(xlle_donor_service_probe C CXX)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  message(FATAL_ERROR "Reference probes must be configured separately from product builds")
endif()

# Separate reference executable: never included by the product build.
set(PHOBOS_SOURCE "" CACHE PATH "External pinned Phobos checkout")
if(NOT WIN32 OR NOT EXISTS "${PHOBOS_SOURCE}/src/hle/audio_output_xaudio2.cpp")
  message(FATAL_ERROR "This private reference probe needs Windows and PHOBOS_SOURCE")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PHOBOS_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE donor_head OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT donor_head STREQUAL "bcccf8cc81d5d2becafe6138dbe81d0f9836b9a0")
  message(FATAL_ERROR "Unreviewed donor revision: ${donor_head}")
endif()
message(STATUS "Private donor qualification only; no product integration or install target")
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
set(XBOXRECOMP_VULKAN OFF CACHE BOOL "Probe the available D3D11 backend" FORCE)
add_subdirectory("${PHOBOS_SOURCE}/src/platform" donor/platform)
add_subdirectory("${PHOBOS_SOURCE}/src/config" donor/config)
add_subdirectory("${PHOBOS_SOURCE}/src/d3d" donor/d3d)

add_executable(donor_renderer_probe renderer.c "${PHOBOS_SOURCE}/src/kernel/nv2a_vsh.c")
target_link_libraries(donor_renderer_probe PRIVATE xbox_d3d8 user32)
add_executable(donor_audio_probe audio.cpp "${PHOBOS_SOURCE}/src/hle/audio_output_xaudio2.cpp")
target_include_directories(donor_audio_probe PRIVATE "${PHOBOS_SOURCE}/src/hle")
target_link_libraries(donor_audio_probe PRIVATE xaudio2 ole32)
enable_testing()
add_test(NAME donor_renderer_pixels COMMAND donor_renderer_probe)
add_test(NAME donor_audio_consumption COMMAND donor_audio_probe)
set_tests_properties(donor_renderer_pixels donor_audio_consumption PROPERTIES TIMEOUT 20)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/display.conf" "")
set_tests_properties(donor_renderer_pixels PROPERTIES ENVIRONMENT
  "RECOMP_DISPLAY_CONFIG=${CMAKE_CURRENT_BINARY_DIR}/display.conf;RECOMP_SHADER_CACHE=0;RECOMP_D3D8_BACKEND=d3d11")
set_tests_properties(donor_audio_consumption PROPERTIES ENVIRONMENT "RECOMP_AUDIO_GAIN=1")
