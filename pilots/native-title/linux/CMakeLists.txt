cmake_minimum_required(VERSION 3.20)
project(xboxllr_native_posix C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This target implements the native Linux host")
endif()
set(PILOT_DXVK_NATIVE "" CACHE PATH "DXVK Native SDK prefix (containing include/dxvk)")
find_path(PILOT_DXVK_INCLUDE windows_base.h HINTS "${PILOT_DXVK_NATIVE}/include/dxvk" REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(pilot_host_posix STATIC host_posix.c host_text.c host_hash.c host_files.c host_exports.c)
target_include_directories(pilot_host_posix PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PILOT_DXVK_INCLUDE}")
target_compile_features(pilot_host_posix PUBLIC c_std_11)
target_compile_options(pilot_host_posix PRIVATE -Wall -Wextra -Werror=implicit-function-declaration)
target_link_libraries(pilot_host_posix PUBLIC Threads::Threads OpenSSL::Crypto ${CMAKE_DL_LIBS})
add_executable(pilot_host_posix_test ../../../tests/test_native_host_posix.c)
target_link_libraries(pilot_host_posix_test PRIVATE pilot_host_posix)
target_compile_options(pilot_host_posix_test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME native_host_posix COMMAND pilot_host_posix_test)
set_tests_properties(native_host_posix PROPERTIES TIMEOUT 20)
