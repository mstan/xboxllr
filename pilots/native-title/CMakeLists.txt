cmake_minimum_required(VERSION 3.20)
project(xlle_native_title_pilot C)
message(STATUS "Private GPL-dependent reference pilot; excluded from product build/install")
set(CMAKE_C_STANDARD 11)
set(PILOT_GENERATED "" CACHE PATH "Private generated title directory")
if(NOT EXISTS "${PILOT_GENERATED}/pilot_image.h")
  message(FATAL_ERROR "Run tools/native_pilot.py and set PILOT_GENERATED to its gen directory")
endif()
file(GLOB generated CONFIGURE_DEPENDS "${PILOT_GENERATED}/*.c")
set(PILOT_MEDIA "OPTICAL" CACHE STRING "Build-time media policy: OPTICAL or INSTALLED")
set_property(CACHE PILOT_MEDIA PROPERTY STRINGS OPTICAL INSTALLED)
if(PILOT_MEDIA STREQUAL "INSTALLED")
  if(NOT EXISTS "${PILOT_GENERATED}/pilot_media_binding.c")
    message(FATAL_ERROR "Installed media requires a reviewed, hash-bound XAPI media helper")
  endif()
  set_source_files_properties(runtime.c "${PILOT_GENERATED}/pilot_media_binding.c"
    PROPERTIES COMPILE_DEFINITIONS PILOT_MEDIA_INSTALLED)
elseif(NOT PILOT_MEDIA STREQUAL "OPTICAL")
  message(FATAL_ERROR "PILOT_MEDIA must be OPTICAL or INSTALLED")
endif()
set(PILOT_CACHE "ORIGINAL" CACHE STRING "Build-time utility drive: ORIGINAL or NATIVE")
set_property(CACHE PILOT_CACHE PROPERTY STRINGS ORIGINAL NATIVE)
if(PILOT_CACHE STREQUAL "NATIVE")
  if(NOT EXISTS "${PILOT_GENERATED}/pilot_cache_binding.c")
    message(FATAL_ERROR "Native cache requires a reviewed, hash-bound XMountUtilityDrive")
  endif()
  set_source_files_properties("${PILOT_GENERATED}/pilot_cache_binding.c"
    PROPERTIES COMPILE_DEFINITIONS PILOT_CACHE_NATIVE)
elseif(NOT PILOT_CACHE STREQUAL "ORIGINAL")
  message(FATAL_ERROR "PILOT_CACHE must be ORIGINAL or NATIVE")
endif()
add_executable(xlle_native_title runtime.c ${generated})
target_include_directories(xlle_native_title PRIVATE "${PILOT_GENERATED}")
target_compile_definitions(xlle_native_title PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(xlle_native_title PRIVATE bcrypt)
enable_testing()
add_executable(pilot_contract_test runtime_contract_test.c)
target_include_directories(pilot_contract_test PRIVATE "${PILOT_GENERATED}")
target_compile_definitions(pilot_contract_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(pilot_contract_test PRIVATE bcrypt)
set(PILOT_D3D_SOURCE "" CACHE PATH "Prepared pinned MIT D3D11 donor directory")
if(PILOT_D3D_SOURCE)
  if(NOT EXISTS "${PILOT_D3D_SOURCE}/donor.json")
    message(FATAL_ERROR "Run tools/native_d3d_donor.py to prepare the backend")
  endif()
  add_library(pilot_d3d_backend STATIC graphics_backend.c
    "${PILOT_D3D_SOURCE}/d3d8_device.c" "${PILOT_D3D_SOURCE}/d3d8_gamma.c"
    "${PILOT_D3D_SOURCE}/d3d8_resources.c" "${PILOT_D3D_SOURCE}/d3d8_shaders.c"
    "${PILOT_D3D_SOURCE}/d3d8_states.c" "${PILOT_D3D_SOURCE}/d3d8_combiners.c"
    "${PILOT_D3D_SOURCE}/d3d8_vsh.c")
  target_include_directories(pilot_d3d_backend PUBLIC "${PILOT_D3D_SOURCE}")
  target_compile_definitions(pilot_d3d_backend PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(pilot_d3d_backend PUBLIC d3d11 dxgi dxguid d3dcompiler)
  add_executable(pilot_graphics_test graphics_backend_test.c)
  target_link_libraries(pilot_graphics_test PRIVATE pilot_d3d_backend)
  add_test(NAME pilot_graphics_backend COMMAND pilot_graphics_test)
  set_tests_properties(pilot_graphics_backend PROPERTIES TIMEOUT 30)
  if(EXISTS "${PILOT_GENERATED}/pilot_d3d_bindings.h")
    target_compile_definitions(pilot_contract_test PRIVATE PILOT_D3D_NATIVE)
    target_link_libraries(pilot_contract_test PRIVATE pilot_d3d_backend)
    add_test(NAME pilot_graphics_abi COMMAND pilot_contract_test graphics)
    add_test(NAME pilot_graphics_unsupported COMMAND pilot_contract_test graphics-unsupported)
    set_tests_properties(pilot_graphics_abi PROPERTIES TIMEOUT 30)
    set_tests_properties(pilot_graphics_unsupported PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
  endif()
endif()
set(PILOT_RENDERER "ORIGINAL" CACHE STRING "Build-time graphics: ORIGINAL or D3D11")
set_property(CACHE PILOT_RENDERER PROPERTY STRINGS ORIGINAL D3D11)
if(PILOT_RENDERER STREQUAL "D3D11")
  if(NOT TARGET pilot_d3d_backend OR NOT EXISTS "${PILOT_GENERATED}/pilot_d3d_binding.c")
    message(FATAL_ERROR "D3D11 needs the prepared donor and reviewed graphics binding")
  endif()
  set_property(SOURCE runtime.c "${PILOT_GENERATED}/pilot_d3d_binding.c"
    APPEND PROPERTY COMPILE_DEFINITIONS PILOT_D3D_NATIVE)
  target_link_libraries(xlle_native_title PRIVATE pilot_d3d_backend)
elseif(NOT PILOT_RENDERER STREQUAL "ORIGINAL")
  message(FATAL_ERROR "PILOT_RENDERER must be ORIGINAL or D3D11")
endif()
add_test(NAME pilot_native_contract COMMAND pilot_contract_test)
add_test(NAME pilot_timer_contract COMMAND pilot_contract_test timers)
add_test(NAME pilot_memory_contract COMMAND pilot_contract_test memory)
add_test(NAME pilot_media_contract COMMAND pilot_contract_test media)
add_test(NAME pilot_storage_contract COMMAND pilot_contract_test storage)
add_test(NAME pilot_section_contract COMMAND pilot_contract_test sections)
add_test(NAME pilot_cache_contract COMMAND pilot_contract_test cache)
add_test(NAME pilot_file_cache_contract COMMAND pilot_contract_test file-cache)
set_tests_properties(pilot_native_contract pilot_timer_contract pilot_memory_contract pilot_media_contract pilot_storage_contract pilot_section_contract pilot_cache_contract pilot_file_cache_contract PROPERTIES TIMEOUT 20)
foreach(kind instruction indirect kernel)
  add_test(NAME pilot_stops_${kind} COMMAND pilot_contract_test ${kind})
  set_tests_properties(pilot_stops_${kind} PROPERTIES WILL_FAIL TRUE)
endforeach()
if(MSVC)
  # Bring-up iteration: donor discovery can produce enormous overlapping
  # functions. Qualify execution before spending minutes optimizing those.
  target_compile_options(xlle_native_title PRIVATE /bigobj /we4013 /Od /MP4)
  target_link_options(xlle_native_title PRIVATE /STACK:8388608)
else()
  target_compile_options(xlle_native_title PRIVATE -O0 -Werror=implicit-function-declaration)
  target_link_options(xlle_native_title PRIVATE -Wl,--stack,8388608)
endif()
