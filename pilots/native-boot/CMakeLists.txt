cmake_minimum_required(VERSION 3.20)
project(xboxllr_native_boot LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BOOT_CORE "${CMAKE_CURRENT_LIST_DIR}/../../src/third_party/xbox-boot-re/core")
add_library(xboxllr_boot_core STATIC
  ${BOOT_CORE}/src/core.c ${BOOT_CORE}/src/pack.c
  ${BOOT_CORE}/src/Timing.c ${BOOT_CORE}/src/x87.c
  ${BOOT_CORE}/src/Overview.c ${BOOT_CORE}/src/Camera.c
  ${BOOT_CORE}/src/Chamber.c ${BOOT_CORE}/src/Blob.c
  ${BOOT_CORE}/src/Logo.c ${BOOT_CORE}/src/Textures.c
  ${BOOT_CORE}/src/Effects.c ${BOOT_CORE}/src/FrameOrder.c
  ${BOOT_CORE}/src/Audio.c)
target_include_directories(xboxllr_boot_core PUBLIC ${BOOT_CORE}/include)
option(XBOXLLR_BOOT_APU_ADAPTER "Use shared native VP audio (OFF is a developer-only legacy comparison)" ON)
if(XBOXLLR_BOOT_APU_ADAPTER)
  if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../cmake/ApuRate.cmake")
    message(FATAL_ERROR "The native VP adapter needs the accompanying shared APU source closure")
  endif()
  include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ApuRate.cmake")
  add_library(xboxllr_boot_apu STATIC boot_audio_apu.cpp boot_audio_owner.cpp)
  target_include_directories(xboxllr_boot_apu PUBLIC ${CMAKE_CURRENT_LIST_DIR})
  target_link_libraries(xboxllr_boot_apu PUBLIC xboxllr_boot_core xlle_apu_rate)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(xboxllr_boot_timeline_test boot_timeline_test.cpp)
  add_test(NAME native_boot_timeline COMMAND xboxllr_boot_timeline_test)
  add_executable(xboxllr_boot_after_image_test boot_after_image_test.cpp)
  add_test(NAME native_boot_after_image COMMAND xboxllr_boot_after_image_test)
  add_executable(xboxllr_boot_audio_sequence_test boot_audio_sequence_test.cpp)
  target_include_directories(xboxllr_boot_audio_sequence_test PRIVATE ${BOOT_CORE}/src)
  target_link_libraries(xboxllr_boot_audio_sequence_test PRIVATE xboxllr_boot_core)
  add_test(NAME native_boot_audio_sequence COMMAND xboxllr_boot_audio_sequence_test)
  if(XBOXLLR_BOOT_APU_ADAPTER)
    add_executable(xboxllr_boot_audio_apu_test boot_audio_apu_test.cpp)
    target_link_libraries(xboxllr_boot_audio_apu_test PRIVATE xboxllr_boot_apu)
    add_test(NAME native_boot_audio_apu COMMAND xboxllr_boot_audio_apu_test)
  endif()
endif()
if(MSVC)
  target_compile_options(xboxllr_boot_core PRIVATE /O2 /fp:strict)
  if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(xboxllr_boot_core PRIVATE /FI${CMAKE_CURRENT_LIST_DIR}/boot_msvc_math.h)
  endif()
else()
  target_compile_options(xboxllr_boot_core PRIVATE -O2 -ffp-contract=off -fno-fast-math)
endif()

# The portable core has no OS, audio, graphics, assembly or file dependencies.
# This first presentation host is Windows only; no other target imports Win32.
if(WIN32)
  add_executable(xboxllr_boot_intro boot_main.cpp boot_renderer.cpp)
  target_compile_definitions(xboxllr_boot_intro PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(xboxllr_boot_intro PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/W4;/O2;/fp:strict>)
  target_link_libraries(xboxllr_boot_intro PRIVATE xboxllr_boot_core d3d11 d3dcompiler dxgi winmm bcrypt user32 gdi32)
  if(XBOXLLR_BOOT_APU_ADAPTER)
    target_link_libraries(xboxllr_boot_intro PRIVATE xboxllr_boot_apu)
    target_compile_definitions(xboxllr_boot_intro PRIVATE XBOXLLR_BOOT_NATIVE_AUDIO=1)
  endif()
endif()
