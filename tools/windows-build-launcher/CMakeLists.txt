cmake_minimum_required(VERSION 3.20)
project(xboxllr_build_tool LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The packaged build-tool entry point targets Windows x64")
endif()
add_executable(xboxllr main.c)
if(MSVC)
  set_target_properties(xboxllr PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
elseif(MINGW)
  target_link_options(xboxllr PRIVATE -municode)
endif()
